Function loco_bin(ByVal n As Integer, ByVal d As Integer) As String
  Dim loco_bs As String
  Dim loco_bi As Integer
  loco_bs = ""
  For loco_bi = 0 To 15
    If n And 1 Then loco_bs = "1" & loco_bs Else loco_bs = "0" & loco_bs
    n = (n \ 2) And &H7FFF
  Next
  While Len(loco_bs) > 1 And Left$(loco_bs, 1) = "0" And Len(loco_bs) > d
    loco_bs = Mid$(loco_bs, 2)
  Wend
  loco_bin = loco_bs
End Function
